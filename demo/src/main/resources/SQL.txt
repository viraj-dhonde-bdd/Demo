CREATE TABLE APP_USERS (
    ID       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    EMAIL    VARCHAR2(255) NOT NULL UNIQUE,
    PASSWORD VARCHAR2(255) NOT NULL,
    ROLE     VARCHAR2(50)  NOT NULL
);

CREATE TABLE REFRESH_TOKENS (
    ID          NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    TOKEN       VARCHAR2(500) NOT NULL UNIQUE,
    USER_ID     NUMBER NOT NULL,
    EXPIRY_DATE TIMESTAMP NOT NULL,

    CONSTRAINT FK_REFRESH_TOKEN_USER
        FOREIGN KEY (USER_ID)
        REFERENCES APP_USERS(ID)
);

CREATE TABLE menus (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR2(100) NOT NULL UNIQUE,
    url VARCHAR2(255),
    icon VARCHAR2(50),
    parent_id NUMBER,
    CONSTRAINT fk_menu_parent FOREIGN KEY (parent_id) REFERENCES menus(id) ON DELETE CASCADE
);

CREATE TABLE bo_menu (
    menu_id INT GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    role VARCHAR(50) NOT NULL,
    title VARCHAR(255) NOT NULL,
    url VARCHAR(255) NOT NULL,
    icon VARCHAR(100) NOT NULL
);

CREATE TABLE bo_submenu (
    sm_id INT GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    menu_id INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    url VARCHAR(255) NOT NULL,
    icon VARCHAR(100) NOT NULL,
    CONSTRAINT fk_submenu_menu FOREIGN KEY (menu_id) REFERENCES bo_menu(menu_id) ON DELETE CASCADE
);
