CREATE TABLE bo_menu (
    menu_id INT GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    role VARCHAR(50) NOT NULL,
    title VARCHAR(255) NOT NULL,
    url VARCHAR(255) NOT NULL,
    icon VARCHAR(100) NOT NULL
);

CREATE TABLE bo_submenu (
    sm_id INT GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    menu_id INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    url VARCHAR(255) NOT NULL,
    icon VARCHAR(100) NOT NULL,
    CONSTRAINT fk_submenu_menu FOREIGN KEY (menu_id) 
    REFERENCES bo_menu(menu_id) ON DELETE CASCADE -- if menu is deleted, delete its submenus as well
);

-- 1. PRODUCTS TABLE
-- Stores items listed by specific vendors
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    vendor_id INT NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    price DECIMAL(10, 2) NOT NULL,
    stock_quantity INT NOT NULL DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (vendor_id) REFERENCES vendors(id) ON DELETE CASCADE
);

-- 2. ORDERS TABLE
-- Stores high-level checkout information for a user from a vendor
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    vendor_id INT NOT NULL,
    total_amount DECIMAL(10, 2) NOT NULL,
    status ENUM('pending', 'processing', 'completed', 'cancelled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (vendor_id) REFERENCES vendors(id) ON DELETE CASCADE
);

-- 3. ORDER_ITEMS TABLE
-- Junction table mapping products to a specific order
CREATE TABLE order_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL,
    price_at_purchase DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE RESTRICT
);

-- CART_ITEMS TABLE
-- Stores items temporarily added to a user's shopping cart before checkout
CREATE TABLE cart_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE,
    -- Ensures a user only has one row per product in their cart (updates quantity instead of duplicating rows)
    UNIQUE KEY unique_user_product (user_id, product_id)
);

JPA Repositories (JpaRepository<Product, Long>): Use these in your controllers/services for quick lookups like productRepository.findById(id) or productRepository.findByVendorId(vendorId).

JdbcTemplate: Use this for complex multi-table joins (like the cart checkout routine above) or when optimizing bulk inserts where JPA's entity tracking overhead can slow things down.
